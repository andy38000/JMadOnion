## 二、准备训练数据

### 2.1 数据来源

需要已经绑定好蒙皮权重的3ds Max文件（.max格式）

### 2.2 单个文件导出

1. 在 3ds Max 中打开已蒙皮的模型
2. 选择蒙皮后的网格模型
3. 菜单：MAXScript → 运行脚本
4. 选择 `export_skinning_simple.py`
5. 数据会保存到指定文件夹

**export_skinning_simple.py 内容：**
```python
# -*- coding: utf-8 -*-
import pymxs
import json
import os

rt = pymxs.runtime

# ===== 修改这里的输出路径 =====
OUTPUT_FOLDER = r"D:\training_data\train"
# =============================

def export_skinning_data():
    sel = rt.selection
    if sel.count == 0:
        print("请先选择一个蒙皮网格!")
        return
    
    mesh_obj = sel[0]
    skin_mod = None
    
    for i in range(1, mesh_obj.modifiers.count + 1):
        mod = mesh_obj.modifiers[i]
        if rt.classOf(mod) == rt.Skin:
            skin_mod = mod
            break
    
    if skin_mod is None:
        print("选中的对象没有Skin修改器!")
        return
    
    # 确保输出目录存在
    if not os.path.exists(OUTPUT_FOLDER):
        os.makedirs(OUTPUT_FOLDER)
    
    rt.select(mesh_obj)
    rt.modPanel.setCurrentObject(skin_mod)
    
    num_verts = rt.skinOps.getNumberVertices(skin_mod)
    num_bones = rt.skinOps.getNumberBones(skin_mod)
    
    print(f"顶点数: {num_verts}, 骨骼数: {num_bones}")
    
    # 获取顶点位置
    vertices = []
    mesh = rt.snapshotAsMesh(mesh_obj)
    for i in range(1, num_verts + 1):
        pos = rt.getVert(mesh, i)
        norm = rt.getNormal(mesh, i) if rt.getNormal(mesh, i) else rt.Point3(0, 0, 1)
        vertices.append({
            "position": [float(pos.x), float(pos.y), float(pos.z)],
            "normal": [float(norm.x), float(norm.y), float(norm.z)]
        })
    
    # 获取骨骼信息
    bones = []
    for i in range(1, num_bones + 1):
        bone_name = rt.skinOps.getBoneName(skin_mod, i, 0)
        bone_node = rt.getNodeByName(bone_name)
        if bone_node:
            head = bone_node.transform.position
            if hasattr(bone_node, 'children') and bone_node.children.count > 0:
                tail = bone_node.children[0].transform.position
            else:
                tail = head + rt.Point3(0, 0, 10)
            bones.append({
                "name": bone_name,
                "head": [float(head.x), float(head.y), float(head.z)],
                "tail": [float(tail.x), float(tail.y), float(tail.z)]
            })
    
    # 获取权重
    weights = []
    for v in range(1, num_verts + 1):
        num_assigned = rt.skinOps.getVertexWeightCount(skin_mod, v)
        vert_weights = {}
        for w in range(1, num_assigned + 1):
            bone_id = rt.skinOps.getVertexWeightBoneID(skin_mod, v, w)
            weight = rt.skinOps.getVertexWeight(skin_mod, v, w)
            bone_name = rt.skinOps.getBoneName(skin_mod, bone_id, 0)
            vert_weights[bone_name] = float(weight)
        weights.append(vert_weights)
    
    # 保存数据
    data = {
        "mesh_name": mesh_obj.name,
        "vertices": vertices,
        "bones": bones,
        "weights": weights
    }
    
    output_file = os.path.join(OUTPUT_FOLDER, f"{mesh_obj.name}_skinning.json")
    with open(output_file, 'w', encoding='utf-8') as f:
        json.dump(data, f, indent=2, ensure_ascii=False)
    
    print(f"导出完成: {output_file}")

export_skinning_data()
```

### 5.2 使用方法

1. 运行 GoSkinning 插件

**方法二：运行独立脚本**

在 3ds Max 中运行以下脚本：

**max_auto_skin.py:**
```python
# -*- coding: utf-8 -*-
import pymxs
import torch
import sys
import os

rt = pymxs.runtime

# ===== 设置模型路径 =====
MODEL_PATH = r"C:\Users\Admin\AppData\Local\Autodesk\3dsMax\2022\ENU\scripts\GoSkinning\models\skinning_model.pt"
# ========================

def auto_skin():
    # 检查选择
    sel = list(rt.selection)
    if len(sel) < 2:
        print("请选择: 1个网格 + 多个骨骼")
        return
    
    # 分离网格和骨骼
    mesh_obj = None
    bones = []
    
    for obj in sel:
        if rt.classOf(obj) == rt.Editable_mesh or rt.classOf(obj) == rt.Editable_Poly:
            mesh_obj = obj
        elif rt.classOf(obj) == rt.BoneGeometry or rt.classOf(obj) == rt.Biped_Object:
            bones.append(obj)
    
    if mesh_obj is None:
        print("未找到网格对象!")
        return
    
    if len(bones) == 0:
        print("未找到骨骼!")
        return
    
    print(f"网格: {mesh_obj.name}, 骨骼数: {len(bones)}")
    
    # 加载模型
    model = torch.jit.load(MODEL_PATH)
    model.eval()
    
    # 获取顶点数据
    mesh = rt.snapshotAsMesh(mesh_obj)
    num_verts = rt.getNumVerts(mesh)
    
    vertices = []
    normals = []
    for i in range(1, num_verts + 1):
        pos = rt.getVert(mesh, i)
        norm = rt.getNormal(mesh, i) if rt.getNormal(mesh, i) else rt.Point3(0, 0, 1)
        vertices.append([pos.x, pos.y, pos.z])
        normals.append([norm.x, norm.y, norm.z])
    
    # 获取骨骼数据
    bone_heads = []
    bone_tails = []
    for bone in bones:
        head = bone.transform.position
        if bone.children.count > 0:
            tail = bone.children[0].transform.position
        else:
            tail = head + rt.Point3(0, 0, 10)
        bone_heads.append([head.x, head.y, head.z])
        bone_tails.append([tail.x, tail.y, tail.z])
    
    # 转换为张量
    v_pos = torch.tensor(vertices, dtype=torch.float32).unsqueeze(0)
    v_norm = torch.tensor(normals, dtype=torch.float32).unsqueeze(0)
    b_heads = torch.tensor(bone_heads, dtype=torch.float32).unsqueeze(0)
    b_tails = torch.tensor(bone_tails, dtype=torch.float32).unsqueeze(0)
    
    # 预测权重
    with torch.no_grad():
        weights = model.predict(v_pos, v_norm, b_heads, b_tails)
    
    weights = weights.squeeze(0).numpy()
    
    # 创建Skin修改器
    skin_mod = rt.Skin()
    rt.addModifier(mesh_obj, skin_mod)
    rt.modPanel.setCurrentObject(skin_mod)
    
    # 添加骨骼
    for bone in bones:
        rt.skinOps.addBone(skin_mod, bone, 0)
    
    # 设置权重
    rt.skinOps.setVertexWeightsEnable(skin_mod, True)
    
    for v_idx in range(num_verts):
        bone_ids = []
        weight_vals = []
        
        for b_idx in range(len(bones)):
            w = weights[v_idx, b_idx]
            if w > 0.01:
                bone_ids.append(b_idx + 1)
                weight_vals.append(w)
        
        if bone_ids:
            rt.skinOps.setVertexWeights(skin_mod, v_idx + 1, bone_ids, weight_vals)
    
    print("自动蒙皮完成!")

auto_skin()
```

## 六、常见问题

### Q1: CUDA out of memory
**解决：** 减小 batch_size 到 1 或 2

### Q2: RuntimeError: selected index k out of range
**解决：** 更新 skinning_net.py 中的 knn 函数（添加 k = min(k, N) 检查）

### Q3: No module named 'torch'
**解决：** 安装 PyTorch：
```cmd
py -3.11 -m pip install torch torchvision --index-url https://download.pytorch.org/whl/cu118
```

### Q4: 训练效果不好
**解决：**
- 增加训练数据量（建议100+个模型）
- 增加训练轮数（epochs）
- 确保训练数据质量（蒙皮权重要准确）
